\begin{descriptionlist}[font=\bodyfont\bfseries]
\item[\href{https://musicpediaa.netlify.app/}{Musicpedia}]
\descriptionstyle{A website where you can search for the lyrics and metadata of songs, albums or artists. Utilizing Deezer API and Genius API for the data. The backend is a simple NodeJS server with Redis database to cache the data and the Frontend is made using Angular and State management using a combination of NGRX and Redux-toolkit.}

\item[\href{https://github.com/Abdelrahman-7gab/Currency-Exchanger}{Currency Exchanger}]
\descriptionstyle{A MERN stack project that uses the Fixer API to showcase the conversion rate between different currencies. The database for storing users’ favorites is MongoDB and Registering and Login authentication flows are handled using JWT Authentication.}

\item[\href{https://github.com/Abdelrahman-7gab/Spare-Cart}{Spare Cart}]
\descriptionstyle{An Angular shopping cart frontend built with clean code and efficient data management. Employing service subject pattern and enabling data sharing between tabs, it adopts a fully reactive and declarative data handling approach.}
\end{descriptionlist}
